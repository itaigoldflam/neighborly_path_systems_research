reducted_wheel(R, i):

    n = len(R)
    R_red = (n-1) * [0]   # n-1 len list
    for 0 <= j <= n-2:
        if j < i:
            if (i - j mod n) < R[j]:     # if i between j, j + r[j], so (i,i+1) in that path
                R_red[j] = R[j] - 1
            else:
                R_red[j] = R[j]
        elif j=i:
            R_red[j] = max(R[i]-1, R[i+1])
        elif j > i:
            if (i - (j+1) mod n) < R[j+1]:
                R_red[j] = R[j+1] - 1
            else:
                R_red[j] = R[j+1]
    return R_red

def get_L_from_R(R: List[int]) -> List[int]:
    """Calculates the CCW reach vector L from CW reach vector R."""
    n = len(R)
    L = [0] * n
    for i in range(n):
        reach = R[i]
        for step in range(1, reach + 1):
            target = (i + step) % n
            L[target] = max(L[target], step)
    return L

def is_contained(R, i, (x,y)):
    n = len(R)
    return i-x mod n < R[x] or i-y mod n < R[y]
